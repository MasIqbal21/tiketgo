bubble_sort.c Bahasa C (WASM Target) 
void bubble_sort_tickets(Ticket arr[], int n) {
    for (int i = 0; i < n - 1; i++) {
        bool swapped = false;
        for (int j = 0; j < n - i - 1; j++) {
            // Perbandingan nilai elemen tiket
            if (arr[j].price > arr[j + 1].price) {
                Ticket temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
                swapped = true;
            }
        }
        if (!swapped) break; // Optimasi early exit
    }
}

int linear_search_ticket(Ticket arr[], int n, const char *query) {
    for (int i = 0; i < n; i++) {
        // Pencarian substring pada judul dan provider
        if (strstr(arr[i].title, query) != NULL ||
            strstr(arr[i].provider, query) != NULL) {
            return i; // Indeks elemen ditemukan
        }
    }
    return -1; // Tidak ditemukan
}
